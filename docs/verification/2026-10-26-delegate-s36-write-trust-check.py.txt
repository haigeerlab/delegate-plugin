import json, os, pathlib, shutil, subprocess, sys
helper = pathlib.Path(sys.argv[1]).resolve()
work = pathlib.Path(sys.argv[2]).resolve() / "w"
if work.exists(): shutil.rmtree(work)
work.mkdir()
def git(cwd, *a): subprocess.run(["git", "-C", str(cwd), *a], check=True, capture_output=True)
M = work / "main"; WT = work / "main-wt"
M.mkdir(); git(M, "init", "-q"); (M / "f").write_text("x\n"); git(M, "add", "f")
git(M, "-c", "user.name=p", "-c", "user.email=p@p", "commit", "-qm", "init")
(M / "sub").mkdir(); git(M, "worktree", "add", "-q", str(WT)); (WT / "sub").mkdir()
UNTRUSTED = '\n[projects.%s]\ntrust_level = "untrusted"\n' % json.dumps(str(M))
cases = [("C0 对照：不加覆盖", M, "", False), ("C1 普通仓库根目录", M, "", True), ("C2 普通仓库子目录", M / "sub", "", True),
         ("C3 worktree 子目录", WT / "sub", "", True), ("C4 已有 untrusted 条目", WT / "sub", UNTRUSTED, True)]
for name, cwd, extra_cfg, use_helper in cases:
    home = work / ("home-" + name.split()[0]); home.mkdir()
    base_cfg = 'model = "gpt-6.1-sol"\n' + extra_cfg
    (home / "config.toml").write_text(base_cfg)
    env = {k: v for k, v in os.environ.items() if not k.startswith(("OPENAI", "CODEX"))}
    env.update(CODEX_HOME=str(home), PWD=str(cwd))
    override = []
    if use_helper:
        h = subprocess.run([sys.executable, "-B", str(helper)], cwd=cwd, env=env, capture_output=True, text=True)
        if h.stdout.strip(): override = ["-c", h.stdout.strip()]
    cmd = ["codex", "exec", "-c", 'approval_policy="never"', *override, "--ephemeral", "--sandbox", "workspace-write",
           "--color", "never", "--", "hi"]
    try:
        p = subprocess.run(cmd, cwd=cwd, env=env, stdin=subprocess.DEVNULL, capture_output=True, text=True, timeout=60)
        rc, out = p.returncode, p.stdout + p.stderr
    except subprocess.TimeoutExpired as e:
        rc, out = "超时60s", (e.stdout or b"").decode(errors="replace") + (e.stderr or b"").decode(errors="replace")
    after = (home / "config.toml").read_text()
    added = after[len(base_cfg):].strip() if after.startswith(base_cfg) else "REWRITTEN"
    print(f"{name} | cwd={cwd.relative_to(work)} | 覆盖={override[1] if override else '无'} | rc={rc} 401={'401' in out} | 写入={added or '否'}")
